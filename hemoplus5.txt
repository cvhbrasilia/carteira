$r = New-Object System.Collections.Generic.List[string]
function L($t) { $r.Add([string]$t) }
$udl = 'C:\Users\Public\Nova pasta\HemoplusVet\bin\HEMOPLUSvet.udl'
$cn = New-Object System.Data.OleDb.OleDbConnection ("File Name=" + $udl)
function Q($sql) { $cm = $cn.CreateCommand(); $cm.CommandText = $sql; $cm.CommandTimeout = 60; $da = New-Object System.Data.OleDb.OleDbDataAdapter $cm; $dt = New-Object System.Data.DataTable; [void]$da.Fill($dt); return ,$dt }
function T($titulo, $sql) {
  L "== $titulo"
  try {
    $dt = Q $sql
    L ("  " + (($dt.Columns | ForEach-Object { $_.ColumnName }) -join ' | '))
    foreach ($w in $dt.Rows) { L ("  " + (($w.ItemArray | ForEach-Object { if ($_ -is [datetime]) { $_.ToString('dd/MM/yyyy HH:mm') } elseif ($_ -is [byte[]]) { '(imagem)' } else { ([string]$_).Trim() } }) -join ' | ')) }
  } catch { L "  erro: $($_.Exception.Message)" }
}
